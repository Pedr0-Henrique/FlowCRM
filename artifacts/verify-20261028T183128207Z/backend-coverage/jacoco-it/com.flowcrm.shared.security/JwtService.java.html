package com.flowcrm.shared.security;

import io.jsonwebtoken.*;
import io.jsonwebtoken.security.Keys;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import javax.crypto.SecretKey;
import java.time.Instant;
import java.time.temporal.ChronoUnit;
import java.util.Date;
import java.util.Map;
import java.util.UUID;

@Service
public class JwtService {

    @Value("${app.jwt.secret}")
    private String secret;

    @Value("${app.jwt.access-token-expiration}")
    private long accessTokenExpiration;

    @Value("${app.jwt.refresh-token-expiration}")
    private long refreshTokenExpiration;

    private SecretKey getSigningKey() {
        byte[] keyBytes = secret.getBytes();
        return Keys.hmacShaKeyFor(keyBytes);
    }

    public String generateAccessToken(UUID userId, UUID companyId, String email, String role) {
        return generateAccessToken(userId, companyId, email, role, 0);
    }

    public String generateAccessToken(UUID userId, UUID companyId, String email, String role, long version) {
        return Jwts.builder()
                .subject(userId.toString())
                .id(UUID.randomUUID().toString())
                .claim("tokenType", "access")
                .claim("tokenVersion", version)
                .claims(Map.of(
                        "companyId", companyId.toString(),
                        "email", email,
                        "role", role
                ))
                .issuedAt(Date.from(Instant.now()))
                .expiration(Date.from(Instant.now().plus(accessTokenExpiration, ChronoUnit.MILLIS)))
                .signWith(getSigningKey())
                .compact();
    }

    public String generateRefreshToken(UUID userId) {
        return generateRefreshToken(userId, 0);
    }

    public String generateRefreshToken(UUID userId, long version) {
        return Jwts.builder()
                .subject(userId.toString())
                .id(UUID.randomUUID().toString())
                .claim("tokenType", "refresh")
                .claim("tokenVersion", version)
                .issuedAt(Date.from(Instant.now()))
                .expiration(Date.from(Instant.now().plus(refreshTokenExpiration, ChronoUnit.MILLIS)))
                .signWith(getSigningKey())
                .compact();
    }

    public Claims extractClaims(String token) {
        return Jwts.parser()
                .verifyWith(getSigningKey())
                .build()
                .parseSignedClaims(token)
                .getPayload();
    }

    public Claims requireToken(String token, String type) {
        Claims claims = extractClaims(token);
        if (!type.equals(claims.get("tokenType", String.class)) || claims.getExpiration() == null
                || claims.get("tokenVersion", Long.class) == null) {
            throw new IllegalArgumentException("Invalid token claims");
        }
        UUID.fromString(claims.getSubject());
        UUID.fromString(claims.getId());
        if ("access".equals(type)) {
            UUID.fromString(claims.get("companyId", String.class));
            com.flowcrm.user.Role.valueOf(claims.get("role", String.class));
            if (claims.get("email", String.class) == null) throw new IllegalArgumentException("Missing email");
        }
        return claims;
    }

    public UUID extractUserId(String token) {
        return UUID.fromString(extractClaims(token).getSubject());
    }

    public UUID extractCompanyId(String token) {
        return UUID.fromString(extractClaims(token).get("companyId", String.class));
    }

    public String extractEmail(String token) {
        return extractClaims(token).get("email", String.class);
    }

    public String extractRole(String token) {
        return extractClaims(token).get("role", String.class);
    }

    public boolean isTokenValid(String token) {
        try {
            Claims claims = extractClaims(token);
            return !claims.getExpiration().before(new Date());
        } catch (JwtException | IllegalArgumentException e) {
            return false;
        }
    }

    public boolean isTokenExpired(String token) {
        try {
            return extractClaims(token).getExpiration().before(new Date());
        } catch (JwtException | IllegalArgumentException e) {
            return true;
        }
    }
}
