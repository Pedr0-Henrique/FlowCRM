package com.flowcrm.config;

import org.springframework.context.annotation.Profile;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.annotation.Bean;
import org.springframework.core.env.Environment;
import org.springframework.beans.factory.config.BeanFactoryPostProcessor;

@Configuration
@Profile("benchmark")
public class BenchmarkConfigurationValidator {
    @Bean
    static BeanFactoryPostProcessor benchmarkDatabaseGuard(Environment environment) {
        return beans -> {
            String url = environment.getRequiredProperty("spring.datasource.url");
            if (!url.matches("jdbc:postgresql://(?:localhost|127\\.0\\.0\\.1|postgres)(?::\\d+)?/\\w+_benchmark")) {
                throw new IllegalStateException("Benchmark requires a local disposable database ending in _benchmark");
            }
        };
    }
}
