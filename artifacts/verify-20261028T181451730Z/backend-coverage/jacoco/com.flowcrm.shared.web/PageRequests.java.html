package com.flowcrm.shared.web;

import com.flowcrm.shared.exception.InvalidParameterException;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Sort;

public final class PageRequests {
    private PageRequests() {}

    public static PageRequest create(int page, int size, String sortBy, String direction, Class<?> entity) {
        if (page < 0 || size < 1 || size > 100) {
            throw new InvalidParameterException("Página deve ser não negativa e tamanho entre 1 e 100.");
        }
        if (!"asc".equalsIgnoreCase(direction) && !"desc".equalsIgnoreCase(direction)) {
            throw new InvalidParameterException("Direção de ordenação inválida.");
        }
        try {
            var field = entity.getDeclaredField(sortBy);
            if (!field.isAnnotationPresent(jakarta.persistence.Column.class)
                    && !field.isAnnotationPresent(jakarta.persistence.Id.class)) {
                throw new NoSuchFieldException(sortBy);
            }
        } catch (NoSuchFieldException exception) {
            throw new InvalidParameterException("Campo de ordenação inválido.");
        }
        return PageRequest.of(page, size, Sort.by(Sort.Direction.fromString(direction), sortBy));
    }
}
