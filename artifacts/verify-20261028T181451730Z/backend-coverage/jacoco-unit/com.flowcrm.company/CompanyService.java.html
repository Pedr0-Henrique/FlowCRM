package com.flowcrm.company;

import com.flowcrm.company.dto.CompanyRequest;
import com.flowcrm.company.dto.CompanyResponse;
import com.flowcrm.shared.exception.ConflictException;
import com.flowcrm.shared.exception.ResourceNotFoundException;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;
import java.util.UUID;

@Service
@RequiredArgsConstructor
public class CompanyService {

    private final CompanyRepository companyRepository;
    private final com.flowcrm.user.UserRepository userRepository;

    public List<CompanyResponse> findAll(UUID companyId) {
        return List.of(findById(companyId, companyId));
    }

    public CompanyResponse findById(UUID id, UUID companyId) {
        requireOwnCompany(id, companyId);
        Company company = companyRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException("Empresa não encontrada"));
        return toResponse(company);
    }

    public CompanyResponse findBySlug(String slug, UUID companyId) {
        Company company = companyRepository.findBySlug(slug)
                .filter(item -> item.getId().equals(companyId))
                .orElseThrow(() -> new ResourceNotFoundException("Empresa não encontrada"));
        return toResponse(company);
    }

    @Transactional
    public CompanyResponse create(CompanyRequest request) {
        throw new com.flowcrm.shared.exception.ForbiddenException(
                "Novas empresas devem ser cadastradas com uma conta independente pelo registro.");
    }

    @Transactional
    public CompanyResponse update(UUID id, UUID companyId, CompanyRequest request) {
        requireOwnCompany(id, companyId);
        Company company = companyRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException("Empresa não encontrada"));

        if (!company.getSlug().equals(request.slug()) && companyRepository.existsBySlug(request.slug())) {
            throw new ConflictException("Slug já está em uso");
        }

        company.setName(request.name());
        company.setSlug(request.slug());
        company = companyRepository.save(company);
        return toResponse(company);
    }

    @Transactional
    public void delete(UUID id, UUID companyId) {
        requireOwnCompany(id, companyId);
        if (userRepository.existsByCompanyId(id)) {
            throw new ConflictException("A empresa possui usuários vinculados e não pode ser excluída.");
        }
        if (!companyRepository.existsById(id)) {
            throw new ResourceNotFoundException("Empresa não encontrada");
        }
        companyRepository.deleteById(id);
        companyRepository.flush();
    }

    private void requireOwnCompany(UUID id, UUID companyId) {
        if (!id.equals(companyId)) throw new ResourceNotFoundException("Empresa não encontrada");
    }

    private CompanyResponse toResponse(Company company) {
        return new CompanyResponse(
                company.getId(),
                company.getName(),
                company.getSlug(),
                company.getCreatedAt(),
                company.getUpdatedAt()
        );
    }
}
